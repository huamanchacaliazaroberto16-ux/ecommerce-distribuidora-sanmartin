BEGIN;

DROP TABLE IF EXISTS detalle_pedidos CASCADE;
DROP TABLE IF EXISTS pedidos CASCADE;
DROP TABLE IF EXISTS productos CASCADE;
DROP TABLE IF EXISTS subcategorias CASCADE;
DROP TABLE IF EXISTS categorias CASCADE;
DROP TABLE IF EXISTS usuarios CASCADE;
DROP TABLE IF EXISTS sucursales CASCADE;
DROP TABLE IF EXISTS empresa_contacto CASCADE;

CREATE TABLE empresa_contacto (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre_empresa      VARCHAR(150) NOT NULL,
    email_pedidos       VARCHAR(150) NOT NULL,
    telefono            VARCHAR(30),
    direccion_referencia TEXT,
    compra_minima       NUMERIC(10,2) NOT NULL DEFAULT 100.00,
    moneda              CHAR(3) NOT NULL DEFAULT 'PEN',
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE sucursales (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre              VARCHAR(120) NOT NULL,
    direccion           TEXT,
    telefono            VARCHAR(30),
    horario             TEXT,
    envio_disponible    BOOLEAN NOT NULL DEFAULT TRUE,
    mapa_url            TEXT,
    observaciones       TEXT,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE usuarios (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre              VARCHAR(150) NOT NULL,
    correo              VARCHAR(150) NOT NULL UNIQUE,
    telefono            VARCHAR(20) NOT NULL,
    dni                 VARCHAR(20) NOT NULL UNIQUE,
    contrasena_hash     TEXT NOT NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT chk_usuarios_telefono
        CHECK (telefono ~ '^[0-9]{9,20}$'),
    CONSTRAINT chk_usuarios_dni
        CHECK (dni ~ '^[0-9]{8,20}$')
);

CREATE TABLE categorias (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre              VARCHAR(120) NOT NULL UNIQUE,
    activo              BOOLEAN NOT NULL DEFAULT TRUE,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE subcategorias (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    categoria_id        BIGINT NOT NULL REFERENCES categorias(id) ON DELETE RESTRICT,
    nombre              VARCHAR(120) NOT NULL,
    activo              BOOLEAN NOT NULL DEFAULT TRUE,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT uq_subcategorias UNIQUE (categoria_id, nombre)
);

CREATE TABLE productos (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    subcategoria_id     BIGINT NOT NULL REFERENCES subcategorias(id) ON DELETE RESTRICT,
    cod                 VARCHAR(50) NOT NULL UNIQUE,
    nombre              VARCHAR(150) NOT NULL,
    descripcion         TEXT,
    precio              NUMERIC(10,2) NOT NULL,
    stock               INTEGER NOT NULL DEFAULT 0,
    imagen_nombre       VARCHAR(200),
    activo              BOOLEAN NOT NULL DEFAULT TRUE,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT chk_productos_precio CHECK (precio >= 0),
    CONSTRAINT chk_productos_stock CHECK (stock >= 0)
);

CREATE TABLE pedidos (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id          BIGINT NOT NULL REFERENCES usuarios(id) ON DELETE RESTRICT,
    fecha_pedido        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    estado              VARCHAR(30) NOT NULL DEFAULT 'creado',
    region              VARCHAR(120) NOT NULL,
    provincia           VARCHAR(120) NOT NULL,
    distrito            VARCHAR(120) NOT NULL,
    codigo_postal       VARCHAR(20),
    direccion           TEXT NOT NULL,
    direccion_raw       TEXT,
    total               NUMERIC(12,2) NOT NULL DEFAULT 0,
    CONSTRAINT chk_pedidos_total CHECK (total >= 0),
    CONSTRAINT chk_pedidos_estado CHECK (
        estado IN ('creado', 'pagado', 'preparando', 'enviado', 'entregado', 'cancelado')
    )
);

CREATE TABLE detalle_pedidos (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    pedido_id           BIGINT NOT NULL REFERENCES pedidos(id) ON DELETE CASCADE,
    producto_id         BIGINT NOT NULL REFERENCES productos(id) ON DELETE RESTRICT,
    cantidad            INTEGER NOT NULL,
    precio_unitario     NUMERIC(10,2) NOT NULL,
    subtotal            NUMERIC(12,2) NOT NULL,
    CONSTRAINT chk_detalle_cantidad CHECK (cantidad > 0),
    CONSTRAINT chk_detalle_precio CHECK (precio_unitario >= 0),
    CONSTRAINT chk_detalle_subtotal CHECK (subtotal >= 0),
    CONSTRAINT uq_detalle_pedido_producto UNIQUE (pedido_id, producto_id)
);

CREATE INDEX idx_subcategorias_categoria_id
    ON subcategorias(categoria_id);

CREATE INDEX idx_productos_subcategoria_id
    ON productos(subcategoria_id);

CREATE INDEX idx_productos_nombre_lower
    ON productos ((LOWER(nombre)));

CREATE INDEX idx_pedidos_usuario_fecha
    ON pedidos(usuario_id, fecha_pedido DESC);

CREATE INDEX idx_detalle_pedidos_pedido_id
    ON detalle_pedidos(pedido_id);

CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_usuarios_updated_at
BEFORE UPDATE ON usuarios
FOR EACH ROW
EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_productos_updated_at
BEFORE UPDATE ON productos
FOR EACH ROW
EXECUTE FUNCTION set_updated_at();

CREATE OR REPLACE FUNCTION set_subtotal_detalle()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.subtotal = ROUND((NEW.cantidad * NEW.precio_unitario)::NUMERIC, 2);
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_set_subtotal_detalle
BEFORE INSERT OR UPDATE OF cantidad, precio_unitario
ON detalle_pedidos
FOR EACH ROW
EXECUTE FUNCTION set_subtotal_detalle();

CREATE OR REPLACE FUNCTION recalcular_total_pedido()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
    v_pedido_id BIGINT;
BEGIN
    v_pedido_id := COALESCE(NEW.pedido_id, OLD.pedido_id);

    UPDATE pedidos
    SET total = COALESCE(
        (SELECT SUM(subtotal) FROM detalle_pedidos WHERE pedido_id = v_pedido_id),
        0
    )
    WHERE id = v_pedido_id;

    RETURN NULL;
END;
$$;

CREATE TRIGGER trg_recalcular_total_pedido
AFTER INSERT OR UPDATE OR DELETE
ON detalle_pedidos
FOR EACH ROW
EXECUTE FUNCTION recalcular_total_pedido();
